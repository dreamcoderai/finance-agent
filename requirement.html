""" Project : XDATA API Description : Redis Cache Key Builder Author : Alok Mohanty """ class CacheKeys: """ Generic Redis cache key generator. """ @staticmethod def build(entity: str, **kwargs) -> str: """ Example: CacheKeys.build( "company_contacts", sf_contact_id="0030...", limit=100 ) Returns: company_contacts:limit=100:sf_contact_id=0030... """ parts = [entity] for key in sorted(kwargs.keys()): value = kwargs[key] if value is not None: parts.append(f"{key}={value}") return ":".join(parts)
